add_library(psx_analyzer STATIC psx_analyze/analyzer.cpp psx_analyze/analyzer.h psx_analyze/asset_tools.cpp psx_analyze/asset_tools.h)

add_executable(recomp_gen recomp_gen/main.cpp)
target_link_libraries(recomp_gen PRIVATE personarm_core)
target_compile_features(recomp_gen PRIVATE cxx_std_17)

add_library(p2text_lib STATIC p2text/p2text.cpp p2text/p2text.h)
target_include_directories(p2text_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/p2text)
target_link_libraries(p2text_lib PUBLIC psx_analyzer)
target_compile_features(p2text_lib PUBLIC cxx_std_17)

target_link_libraries(p2text PRIVATE p2text_lib)

add_executable(trace_compare trace_compare/main.cpp)
target_link_libraries(trace_compare PRIVATE personarm_core)

if(MSVC)
    target_compile_options(psx_analyzer PRIVATE /W4)
    target_compile_options(p2text_lib PRIVATE /W4)
    target_compile_options(p2text PRIVATE /W4)
    target_compile_definitions(psx_analyzer PUBLIC _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS NOMINMAX)
else()
    target_compile_options(psx_analyzer PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(p2text_lib PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(p2text PRIVATE -Wall -Wextra -Wpedantic)
endif()

# personarm_native: first-run native code for the player's disc (shipped; docs/NATIVE_CODE.md).
# The compiler is TinyCC (cmake/TinyCC.cmake, fetched at a pinned commit, LGPL, linked as libtcc.dll).
# Without it the tool still emits C for developers (--emit-c) but cannot build a library.
if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(_prm_tcc_default ON)
else()
    set(_prm_tcc_default OFF)
endif()
option(PRM_WITH_TCC "Build personarm_native with TinyCC (fetched at configure time)" ${_prm_tcc_default})
file(READ ${PROJECT_SOURCE_DIR}/src/recomp/runtime/prm_abi.h _prm_abi_text)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/recomp/runtime/prm_abi.h)
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/prm_abi_text.cpp CONTENT
    "// prm_abi.h, embedded for the on-device compiler (generated by CMake).\nnamespace personarm::native {\nextern const char kPrmAbiHeader[] = R\"PRMABI(${_prm_abi_text})PRMABI\";\n}\n")
add_library(native_prep STATIC personarm_native/native_prep.cpp personarm_native/native_prep.h ${CMAKE_CURRENT_BINARY_DIR}/prm_abi_text.cpp)
target_include_directories(native_prep PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/personarm_native)
target_link_libraries(native_prep PUBLIC psx_analyzer personarm_core)
target_compile_features(native_prep PUBLIC cxx_std_17)
if(PRM_WITH_TCC)
    include(${PROJECT_SOURCE_DIR}/cmake/TinyCC.cmake)
    target_link_libraries(native_prep PRIVATE libtcc)
    target_compile_definitions(native_prep PRIVATE PRM_HAVE_TCC PRM_TCC_VERSION="${_tcc_version}")
endif()
add_executable(personarm_native personarm_native/main.cpp)
target_link_libraries(personarm_native PRIVATE native_prep)
install(TARGETS personarm_native RUNTIME DESTINATION bin)
